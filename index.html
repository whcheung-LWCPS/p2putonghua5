        function base64ToArrayBuffer(base64) {

            const binaryString = window.atob(base64);

            const len = binaryString.length;

            const bytes = new Uint8Array(len);

            for (let i = 0; i < len; i++) {

                bytes[i] = binaryString.charCodeAt(i);

            }

            return bytes.buffer;

        }


        function pcmToWav(pcm16Data, sampleRate) {

            const numChannels = 1;

            const bitsPerSample = 16;

            const byteRate = sampleRate * numChannels * (bitsPerSample / 8);

            const blockAlign = numChannels * (bitsPerSample / 8);

            const dataSize = pcm16Data.length * 2;

            const chunkSize = 36 + dataSize;

            

            const buffer = new ArrayBuffer(44 + dataSize);

            const view = new DataView(buffer);

            

            writeString(view, 0, 'RIFF');

            view.setUint32(4, chunkSize, true);

            writeString(view, 8, 'WAVE');

            

            writeString(view, 12, 'fmt ');

            view.setUint32(16, 16, true);

            view.setUint16(20, 1, true);

            view.setUint16(22, numChannels, true);

            view.setUint32(24, sampleRate, true);

            view.setUint32(28, byteRate, true);

            view.setUint16(32, blockAlign, true);

            view.setUint16(34, bitsPerSample, true);

            

            writeString(view, 36, 'data');

            view.setUint32(40, dataSize, true);

            

            let offset = 44;

            for (let i = 0; i < pcm16Data.length; i++, offset += 2) {

                view.setInt16(offset, pcm16Data[i], true);

            }

            

            return new Blob([buffer], { type: 'audio/wav' });

        }


        function writeString(view, offset, string) {

            for (let i = 0; i < string.length; i++) {

                view.setUint8(offset + i, string.charCodeAt(i));

            }

        }

                            responseModalities: ["AUDIO"],

                            speechConfig: {

                                voiceConfig: {

                                    prebuiltVoiceConfig: { voiceName: "Kore" }

                                }

                            }

                        }

                    };


                    const response = await fetch(apiUrl, {

                        method: 'POST',

                        headers: { 'Content-Type': 'application/json' },

                        body: JSON.stringify(payload)

                    });


                    if (response.ok) {

                        const result = await response.json();

                        const part = result?.candidates?.[0]?.content?.parts?.[0];

                        const audioData = part?.inlineData?.data;

                        const mimeType = part?.inlineData?.mimeType || "audio/L16;rate=24000";


                        if (audioData) {

                            let sampleRate = 24000;

                            const match = mimeType.match(/rate=(\d+)/);

                            if (match && match[1]) sampleRate = parseInt(match[1], 10);

                            

                            const pcmData = base64ToArrayBuffer(audioData);

                            const pcm16 = new Int16Array(pcmData);

                            const wavBlob = pcmToWav(pcm16, sampleRate);

                            const audioUrl = URL.createObjectURL(wavBlob);


                            audioCache[text] = audioUrl;

                            currentAudioPlayer = new Audio(audioUrl);

                            currentAudioPlayer.play();

                            return;

                        }

                    }

                }

            } catch (err) {}


            // 2. 嘗試 Google 國語/普通話語音接口 (適合 GitHub Pages，無需 API Key，支援長句)

            try {

                const googleMandarinUrl = `https://translate.google.com/translate_tts?ie=UTF-8&q=${encodeURIComponent(text)}&tl=zh-CN&client=tw-ob`;

                currentAudioPlayer = new Audio(googleMandarinUrl);

                await currentAudioPlayer.play();

                audioCache[text] = googleMandarinUrl;

                return;

            } catch (e) {

                // 3. 次要備用：有道語音

                try {

                    const onlineMandarinUrl = `https://dict.youdao.com/dictvoice?audio=${encodeURIComponent(text)}&type=1`;

                    currentAudioPlayer = new Audio(onlineMandarinUrl);

                    await currentAudioPlayer.play();

                    return;

                } catch (err2) {

                    // 4. 最終備用：iPad / Web Speech API (已優化 iOS 國語聲線選取)

                    speakTextWebSpeechFallback(text);

                }

            }

        }


        function speakTextWebSpeechFallback(text) {

            const synth = window.speechSynthesis;

            if (!synth) return;

            synth.cancel();


            const voices = synth.getVoices() || [];

            

            // 嚴格篩選 iPad/iOS 上的國語/普通話聲線，排除廣東話 (Sin-Ji 等)

            let mandarinVoice = voices.find(v => {

                const lang = (v.lang || '').toLowerCase();

                const name = (v.name || '').toLowerCase();

                const isCanto = lang.includes('hk') || lang.includes('yue') || name.includes('cantonese') || name.includes('sin-ji') || name.includes('sinji') || name.includes('粵語') || name.includes('廣東話');

                const isMandarin = lang.includes('zh-cn') || lang.includes('zh-tw') || lang.includes('zh-sg') || lang.includes('zh-cmn') || name.includes('putonghua') || name.includes('mandarin') || name.includes('國語') || name.includes('ting-ting') || name.includes('mei-jia') || name.includes('lili') || name.includes('yushu');

                return !isCanto && isMandarin;

            });


            // 若未精準匹配，則選取任意非廣東話的中文聲線

            if (!mandarinVoice) {

                mandarinVoice = voices.find(v => {

                    const lang = (v.lang || '').toLowerCase();

                    const name = (v.name || '').toLowerCase();

                    const isCanto = lang.includes('hk') || lang.includes('yue') || name.includes('cantonese') || name.includes('sin-ji') || name.includes('sinji') || name.includes('粵語') || name.includes('廣東話');

                    return !isCanto && lang.startsWith('zh');

                });

            }


            const utterance = new SpeechSynthesisUtterance(text);

            utterance.lang = 'zh-CN';

            if (mandarinVoice) {

                utterance.voice = mandarinVoice;

            }

            utterance.rate = 0.85;

            synth.speak(utterance);

        }

                if (currentAudioPlayer) currentAudioPlayer.pause();

                if (window.speechSynthesis) window.speechSynthesis.cancel();

            }

        }

        function handleDragStart(e, wordId) {

            e.dataTransfer.setData('text/plain', wordId);

            selectedWordId = wordId;

        }


        function handleDragOver(e) {

            e.preventDefault();

            e.currentTarget.classList.add('drop-zone-active');

        }


        function handleDragLeave(e) {

            e.currentTarget.classList.remove('drop-zone-active');

        }


        function handleDrop(e) {

            e.preventDefault();

            e.currentTarget.classList.remove('drop-zone-active');

            const wordId = e.dataTransfer.getData('text/plain') || selectedWordId;

            const slotType = e.currentTarget.getAttribute('data-type');

            

            if (wordId) {

                placeWordIntoSlot(wordId, slotType);

            }

        }


        function handleCardClick(wordId) {

            document.querySelectorAll('.word-card').forEach(c => c.classList.remove('ring-4', 'ring-blue-400', 'scale-105'));


            if (selectedWordId === wordId) {

                selectedWordId = null;

            } else {

                selectedWordId = wordId;

                const card = document.getElementById(`card-${wordId}`);

                if (card) {

                    card.classList.add('ring-4', 'ring-blue-400', 'scale-105');

                }

            }

        }


        function handleSlotClick(slotType) {

            if (selectedWordId) {

                placeWordIntoSlot(selectedWordId, slotType);

                selectedWordId = null;

                document.querySelectorAll('.word-card').forEach(c => c.classList.remove('ring-4', 'ring-blue-400', 'scale-105'));

            } else if (currentAnswers[slotType]) {

                currentAnswers[slotType] = null;

                updateSlotsUI();

                checkSentenceCompletion();

            }

        }


        function placeWordIntoSlot(wordId, slotType) {

            const word = WORD_DATA.find(w => w.id === wordId);

            if (!word) return;


            if (word.type !== slotType) {

                playSound('wrong');

                return;

            }


            let isCorrect = false;

            if (currentScenario === 'sunny') {

                if (slotType === 'weather' && word.isHot) isCorrect = true;

                if (slotType === 'item' && word.isHot) isCorrect = true;

            } else if (currentScenario === 'rainy') {

                if (slotType === 'weather' && word.isRain) isCorrect = true;

                if (slotType === 'item' && word.isRain) isCorrect = true;

            }


            if (isCorrect) {

                currentAnswers[slotType] = word;

                playSound('correct');

                updateSlotsUI();

                checkSentenceCompletion();

            } else {

                playSound('wrong');

            }

        }

        }
